    <script>
        (function() {
            const btn = document.getElementById('likeBtnApp');
            const countEl = document.getElementById('likeBtnCount');
            if (!btn || !countEl) return;

            let count = 0;

            btn.addEventListener('click', function(e) {
                // 1. カウントアップ
                count++;
                countEl.textContent = count;

                // 2. クリックされたボタンの位置（画面上の絶対座標）を取得
                const rect = btn.getBoundingClientRect();
                const btnX = rect.left + rect.width / 2;
                const btnY = rect.top;

                // 1回のクリックで3個のパーティクルを生成
                for (let i = 0; i < 3; i++) {
                    createParticle(btnX, btnY);
                }
            });

            function createParticle(startX, startY) {
                const particle = document.createElement('div');
                particle.className = 'like-particle';
                particle.textContent = '❤️';
                
                // 初期の左右のブレ・サイズをランダムに設定
                const randomOffset = (Math.random() - 0.5) * 40; // 左右に最大20px散らす
                const randomScale = 0.5 + Math.random() * 0.8; // サイズのばらつき
                
                particle.style.left = (startX + randomOffset) + 'px';
                particle.style.top = startY + 'px';
                particle.style.transform = `scale(${randomScale})`;

                document.body.appendChild(particle);

                // JavaScriptのアニメーションでサイト上部（Webページの一番上）まで移動させる
                let currentY = startY;
                let currentX = startX + randomOffset;
                
                // 昇るスピードと横揺れの周期をランダム化
                const speed = 4 + Math.random() * 4; // 1フレームあたりの上昇ピクセル数
                const swingRange = 1 + Math.random() * 2; // 横揺れの幅
                let angle = Math.random() * 100;

                function animate() {
                    currentY -= speed; // 上へ移動
                    angle += 0.05;
                    const swing = Math.sin(angle) * swingRange; // ふわふわ揺れる動き
                    currentX += swing;

                    // 画面上部に近づくにつれて透明度（opacity）を落とす
                    // （残り距離が少ない、または画面外（Y<=0）に出たら消滅）
                    const opacity = Math.min(1, currentY / (startY * 0.8));
                    
                    particle.style.top = currentY + 'px';
                    particle.style.left = currentX + 'px';
                    particle.style.opacity = opacity;

                    if (currentY > -30 && opacity > 0.01) {
                        requestAnimationFrame(animate);
                    } else {
                        particle.remove(); // 画面上部に到達、または見えなくなったら削除
                    }
                }

                requestAnimationFrame(animate);
            }
        })();
    </script>
</div>

    <script>
        (function() {
            const btn = document.getElementById('fireworkBtn');
            if (!btn) return;

            // 鮮やかな花火のカラーバリエーション
            const colors = [
                '#ff3366', '#00f0ff', '#ffea00', '#39ff14', 
                '#ff00ff', '#ff9900', '#ffffff', '#bf55ec'
            ];

            btn.addEventListener('click', function(e) {
                // 1. ボタンを押した画面上の座標を取得
                const rect = btn.getBoundingClientRect();
                const startX = rect.left + rect.width / 2;
                const startY = rect.top;

                // 2. 打ち上げ目標の高さ（現在の画面の適当な上部エリア）
                const targetY = window.innerHeight * (0.15 + Math.random() * 0.25);

                // 3. 打ち上げ光の玉を生成
                createRocket(startX, startY, targetY);
            });

            // 火花が昇るアニメーション
            function createRocket(startX, startY, targetY) {
                const rocket = document.createElement('div');
                rocket.className = 'fw-particle';
                rocket.style.backgroundColor = '#ffea00';
                rocket.style.boxShadow = '0 0 10px #ffea00, 0 0 20px #ffea00';
                rocket.style.left = startX + 'px';
                rocket.style.top = startY + 'px';
                document.body.appendChild(rocket);

                let currentY = startY;
                const speed = 12; // 上昇スピード

                function launch() {
                    currentY -= speed;
                    rocket.style.top = currentY + 'px';

                    // 目標の高さに達したら爆発
                    if (currentY <= targetY) {
                        rocket.remove();
                        explode(startX, targetY);
                    } else {
                        requestAnimationFrame(launch);
                    }
                }
                requestAnimationFrame(launch);
            }

            // ドカンと広がる爆発アニメーション
            function explode(x, y) {
                const particleCount = 40 + Math.floor(Math.random() * 20); // 火花の数
                const baseColor = colors[Math.floor(Math.random() * colors.length)]; // この花火の色
                const nextColor = colors[Math.floor(Math.random() * colors.length)]; // 変化用の色

                for (let i = 0; i < particleCount; i++) {
                    const p = document.createElement('div');
                    p.className = 'fw-particle';
                    
                    // 2色混ざった綺麗な花火にする
                    p.style.backgroundColor = Math.random() > 0.4 ? baseColor : nextColor;
                    p.style.boxShadow = `0 0 6px ${p.style.backgroundColor}, 0 0 12px ${p.style.backgroundColor}`;
                    p.style.left = x + 'px';
                    p.style.top = y + 'px';
                    document.body.appendChild(p);

                    // 360度ランダムな方向に飛び散る角度と速度を計算
                    const angle = Math.random() * Math.PI * 2;
                    const velocity = 3 + Math.random() * 6; // 飛び散る勢い
                    let vx = Math.cos(angle) * velocity;
                    let vy = Math.sin(angle) * velocity;
                    
                    let px = x;
                    let py = y;
                    let opacity = 1;
                    const gravity = 0.08; // 重力（少しずつ下に垂れる）
                    const friction = 0.96; // 空気抵抗（だんだん減速する）

                    function updateParticle() {
                        // 物理演算の計算
                        vx *= friction;
                        vy *= friction;
                        vy += gravity; 
                        px += vx;
                        py += vy;
                        opacity -= 0.015; // 徐々に消えていく

                        p.style.left = px + 'px';
                        p.style.top = py + 'px';
                        p.style.opacity = opacity;

                        if (opacity > 0) {
                            requestAnimationFrame(updateParticle);
                        } else {
                            p.remove(); // 見えなくなったら削除
                        }
                    }
                    requestAnimationFrame(updateParticle);
                }
            }
        })();
    </script>
</div>
